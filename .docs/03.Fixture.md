# Fixture: Nó là một hàm để khởi tạo dữ liệu cho test. Dùng chung cho tất cả các test.
# POM: Nó là 1 class để tổ chức code của 1 trang web, có locators và actions.

# Khởi tạo Fixture
```ts
import { test as base, expect } from "@playwright/test";

export { expect };

export const test = base.extend<{
  landingPage: Page;
}>({
  landingPage: async ({ page }, use) => {
    // before each 
    // giải sử login....
    const landingPage = new LandingPage(page);
    
    // use test case
    await use(landingPage);

    // after eache
    // giả sử clear data
  },
});

import { test as base, expect } from "@playwright/test";

export { expect };

export const test = base.extend<{
  casepage: Page;
}>({
  casepage: async ({ page }, use) => {
    // before each 
    // giải sử login....
    const casepage = new Casepage(page);
    
    // use test case
    await use(casepage);

    // after eache
    // giả sử clear data
  },
});

```

# Nếu có nhiều fixture, thì cần merge fixture
```ts
// liet ke tat ca cac fixture
import { test as testFixtures } from './testFixtures';
import { caseFixture } from "./case-fixture";
import { mergeTests } from "@playwright/test";

// merge
export const test = mergeTests(caseFixture, testFixtures);

```
